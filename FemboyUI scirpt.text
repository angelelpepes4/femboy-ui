local AllConnections = {}
local AllTweens = {}
local Util = {}
function Util.Create(class, props, children)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then
            inst[k] = v
        end
    end
    for _, child in pairs(children or {}) do
        child.Parent = inst
    end
    if props and props.Parent then
        inst.Parent = props.Parent
    end
    return inst
end
function Util.Tween(inst, props, dur, style, dir)
    local tw = TweenService:Create(
        inst,
        TweenInfo.new(dur or 0.3, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out),
        props
    )
    tw:Play()
    table.insert(AllTweens, tw)
    return tw
end
function Util.Spring(inst, props, dur)
    return Util.Tween(inst, props, dur or 0.5, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
end
function Util.Bounce(inst, props, dur)
    return Util.Tween(inst, props, dur or 0.6, Enum.EasingStyle.Bounce, Enum.EasingDirection.Out)
end
function Util.Elastic(inst, props, dur)
    return Util.Tween(inst, props, dur or 0.7, Enum.EasingStyle.Elastic, Enum.EasingDirection.Out)
end
function Util.Ripple(parent, x, y)
    local circle = Util.Create("Frame", {
        Parent = parent,
        BackgroundColor3 = Color3.new(1, 1, 1),
        BackgroundTransparency = 0.65,
        Position = UDim2.new(0, x - parent.AbsolutePosition.X, 0, y - parent.AbsolutePosition.Y),
        Size = UDim2.new(0, 0, 0, 0),
        AnchorPoint = Vector2.new(0.5, 0.5),
        ZIndex = parent.ZIndex + 10,
    }, {
        Util.Create("UICorner", {CornerRadius = UDim.new(1, 0)})
    })
    local maxDim = math.max(parent.AbsoluteSize.X, parent.AbsoluteSize.Y) * 2.5
    Util.Tween(circle, {Size = UDim2.new(0, maxDim, 0, maxDim), BackgroundTransparency = 1}, 0.65)
    task.delay(0.7, function() if circle then circle:Destroy() end end)
end
function Util.LerpColor(a, b, t)
    return Color3.new(a.R + (b.R - a.R) * t, a.G + (b.G - a.G) * t, a.B + (b.B - a.B) * t)
end
function Util.HSVShift(color, shift)
    local h, s, v = color:ToHSV()
    return Color3.fromHSV((h + shift) % 1, s, v)
end
function Util.Connect(signal, func)
    local conn = signal:Connect(func)
    table.insert(AllConnections, conn)
    return conn
end
function Util.Hex(color)
    return string.format("#%02X%02X%02X", math.floor(color.R * 255), math.floor(color.G * 255), math.floor(color.B * 255))
end
local Sound = {}
Sound.Enabled = true
function Sound.Play(id, vol, pitch)
    if not Sound.Enabled then return end
    pcall(function()
        local s = Instance.new("Sound")
        s.SoundId = id
        s.Volume = vol or 0.2
        s.PlaybackSpeed = pitch or 1
        s.Parent = SoundService
        s:Play()
        s.Ended:Connect(function() s:Destroy() end)
    end)
end

local Config = {}
Config.Flags = {}
Config.FileName = "FemboyUI_Config.json"
Config.AutoSave = true
Config.Callbacks = {}
function Config.Set(key, value)
    Config.Flags[key] = value
    if Config.AutoSave then Config.Save() end
    if Config.Callbacks[key] then
        pcall(Config.Callbacks[key], value)
    end
end
function Config.Get(key, default)
    if Config.Flags[key] ~= nil then return Config.Flags[key] end
    return default
end
function Config.Save()
    pcall(function()
        if writefile then
            writefile(Config.FileName, HttpService:JSONEncode(Config.Flags))
        end
    end)
end
function Config.Load()
    pcall(function()
        if isfile and isfile(Config.FileName) then
            Config.Flags = HttpService:JSONDecode(readfile(Config.FileName))
        end
    end)
end
local Particles = {}
function Particles.Sparkles(parent, count)
    local sparkles = {}
    for i = 1, (count or 18) do
        local sp = Util.Create("Frame", {
            Name = "Sparkle" .. i,
            Parent = parent,
            BackgroundColor3 = CurrentTheme.Primary,
            BackgroundTransparency = math.random(40, 85) / 100,
            Size = UDim2.new(0, math.random(2, 5), 0, math.random(2, 5)),
            Position = UDim2.new(math.random(0, 1000) / 1000, 0, math.random(0, 1000) / 1000, 0),
            AnchorPoint = Vector2.new(0.5, 0.5),
            ZIndex = 2,
        }, {
            Util.Create("UICorner", {CornerRadius = UDim.new(1, 0)})
        })
        table.insert(sparkles, sp)
    end
    task.spawn(function()
        while parent and parent.Parent do
            for _, sp in ipairs(sparkles) do
                if sp and sp.Parent then
                    Util.Tween(sp, {
                        Position = UDim2.new(math.random(0, 1000) / 1000, 0, math.random(0, 1000) / 1000, 0),
                        BackgroundTransparency = math.random(30, 92) / 100,
                        Size = UDim2.new(0, math.random(1, 6), 0, math.random(1, 6)),
                        BackgroundColor3 = Util.HSVShift(CurrentTheme.Primary, math.random(0, 40) / 100),
                        Rotation = math.random(0, 360),
                    }, math.random(18, 40) / 10)
                end
            end
            task.wait(math.random(12, 28) / 10)
        end
    end)
    return sparkles
end

local Rainbow = {}
function Rainbow.AnimatedGradient(frame)
    local grad = Util.Create("UIGradient", {
        Parent = frame,
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, CurrentTheme.GradientStart),
            ColorSequenceKeypoint.new(0.5, CurrentTheme.GradientMid),
            ColorSequenceKeypoint.new(1, CurrentTheme.GradientEnd),
        }),
        Rotation = 0,
    })
    task.spawn(function()
        local rot = 0
        while grad and grad.Parent do
            rot = (rot + 0.4) % 360
            grad.Rotation = rot
            local shift = (tick() * 0.08) % 1
            pcall(function()
                grad.Color = ColorSequence.new({
                    ColorSequenceKeypoint.new(0, Util.HSVShift(CurrentTheme.GradientStart, shift)),
                    ColorSequenceKeypoint.new(0.5, Util.HSVShift(CurrentTheme.GradientMid, shift)),
                    ColorSequenceKeypoint.new(1, Util.HSVShift(CurrentTheme.GradientEnd, shift)),
                })
            end)
            RunService.Heartbeat:Wait()
        end
    end)
    return grad
end
function Rainbow.AnimatedBorder(frame, thickness)
    local stroke = Util.Create("UIStroke", {
        Parent = frame,
        Thickness = thickness or 2,
        Transparency = 0.25,
    })
    local sGrad = Util.Create("UIGradient", {
        Parent = stroke,
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 100, 150)),
            ColorSequenceKeypoint.new(0.17, Color3.fromRGB(255, 255, 100)),
            ColorSequenceKeypoint.new(0.33, Color3.fromRGB(100, 255, 150)),
            ColorSequenceKeypoint.new(0.5, Color3.fromRGB(100, 220, 255)),
            ColorSequenceKeypoint.new(0.67, Color3.fromRGB(150, 100, 255)),
            ColorSequenceKeypoint.new(0.83, Color3.fromRGB(255, 100, 220)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 100, 150)),
        }),
    })
    task.spawn(function()
        local offset = 0
        while stroke and stroke.Parent do
            offset = (offset + 0.002) % 1
            sGrad.Offset = Vector2.new(offset, 0)
            RunService.Heartbeat:Wait()
        end
    end)
    return stroke
end
function Rainbow.PulsingGlow(frame, color)
    local glow = Util.Create("UIStroke", {
        Parent = frame,
        Color = color or CurrentTheme.Glow,
        Thickness = 1.5,
        Transparency = 0.5,
    })
    task.spawn(function()
        while glow and glow.Parent do
            Util.Tween(glow, {Transparency = 0.2, Thickness = 2.5}, 1.5, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
            task.wait(1.5)
            Util.Tween(glow, {Transparency = 0.7, Thickness = 1}, 1.5, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
            task.wait(1.5)
        end
    end)
    return glow
end
local CursorTrail = {}
function CursorTrail.Create(screenGui)
    local trails = {}
    local maxParts = 14
    local positions = {}
    for i = 1, maxParts do
        positions[i] = Vector2.new(0, 0)
        local size = math.max(2, 14 - i)
        local trail = Util.Create("Frame", {
            Name = "Trail" .. i,
            Parent = screenGui,
            BackgroundColor3 = Util.LerpColor(CurrentTheme.GradientStart, CurrentTheme.GradientEnd, (i - 1) / (maxParts - 1)),
            BackgroundTransparency = 0.2 + (i / maxParts) * 0.7,
            Size = UDim2.new(0, size, 0, size),
            AnchorPoint = Vector2.new(0.5, 0.5),
            ZIndex = 99999 - i,
        }, {
            Util.Create("UICorner", {CornerRadius = UDim.new(1, 0)})
        })
        table.insert(trails, trail)
    end
    Util.Connect(RunService.RenderStepped, function()
        positions[1] = Vector2.new(Mouse.X, Mouse.Y)
        for i = #positions, 2, -1 do
            positions[i] = positions[i]:Lerp(positions[i - 1], 0.28)
        end
        for i, trail in ipairs(trails) do
            if trail and trail.Parent then
                trail.Position = UDim2.new(0, positions[i].X, 0, positions[i].Y)
                trail.BackgroundColor3 = Util.HSVShift(
                    Util.LerpColor(CurrentTheme.GradientStart, CurrentTheme.GradientEnd, (i - 1) / (maxParts - 1)),
                    (tick() * 0.15 + i * 0.04) % 1
                )
            end
        end
    end)
    return trails
end

        BackgroundColor3 = CurrentTheme.Background,
        BackgroundTransparency = 0.25,
        Size = UDim2.new(0, 280, 0, 30),
        Position = UDim2.new(0, 15, 0, 15),

    }, {

    local wmLabel = Util.Create("TextLabel", {
        Parent = wmFrame,

        Size = UDim2.new(1, -16, 1, 0),

        Text = "🌸 " .. title,
        TextColor3 = CurrentTheme.Text,
        TextSize = 12,
        Font = Enum.Font.GothamBold,
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex = 40001,
    })
    task.spawn(function()
        local lastTick = tick()
        local fps = 60
        while wmFrame and wmFrame.Parent do
            local now = tick()
            fps = math.floor(1 / (now - lastTick))
            lastTick = now
            local timeStr = os.date("%I:%M:%S %p")
            wmLabel.Text = string.format("🌸 %s  |  ⏰ %s  |  📊 %d FPS", title, timeStr, fps)
            RunService.Heartbeat:Wait()
        end
    end)
    return wmFrame
end
function FemboyUI.new(config)
    config = config or {}
    local self = setmetatable({}, FemboyUI)
    self.Title = config.Title or "✨ Femboy UI ✨"
    self.SubTitle = config.SubTitle or "The Ultimate Experience"
    self.Theme = config.Theme or "Femboy"
    self.WindowSize = config.Size or UDim2.new(0, 640, 0, 460)
    self.EnableCursorTrail = config.CursorTrail ~= false
    self.EnableParticles = config.Particles ~= false
    self.EnableSounds = config.Sounds ~= false
    self.EnableWatermark = config.Watermark ~= false
    self.ConfigName = config.ConfigName or "FemboyUI"
    self.ToggleKey = config.ToggleKey or Enum.KeyCode.RightControl
    self.IsMobile = UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled
    self.Responsive = config.Responsive ~= false
    if Themes[self.Theme] then
        CurrentTheme = Themes[self.Theme]
    end
    Sound.Enabled = self.EnableSounds
    Config.FileName = self.ConfigName .. "_Config.json"
    Config.Load()
    self.Tabs = {}
    self.ActiveTab = nil
    self.Visible = true
    self.Minimized = false
    self:Build()
    return self
end
function FemboyUI:GetResponsiveSize()
    if not self.Responsive then return self.WindowSize end
    local camera = workspace.CurrentCamera
    local viewport = camera and camera.ViewportSize or Vector2.new(1280, 720)
    local width = math.clamp(viewport.X - (self.IsMobile and 18 or 80), 300, self.WindowSize.X.Offset)
    local height = math.clamp(viewport.Y - (self.IsMobile and 90 or 120), 300, self.WindowSize.Y.Offset)
    return UDim2.new(0, width, 0, height)
end
function FemboyUI:ApplyResponsiveLayout()
    if not self.Responsive or not self.MainFrame then return end
    self.MainFrame.Size = self.Visible and self:GetResponsiveSize() or UDim2.new(0, 0, 0, 0)
    if self.IsMobile and self.Sidebar then
        self.Sidebar.Size = UDim2.new(0, 112, 1, -10)
        self.TabContent.Position = UDim2.new(0, 122, 0, 5)
        self.TabContent.Size = UDim2.new(1, -127, 1, -10)
    end
end
function FemboyUI:Build()
    pcall(function()
        local old = GuiParent:FindFirstChild("FemboyUI_Main")
        if old then old:Destroy() end
    end)
    self.ScreenGui = Util.Create("ScreenGui", {
        Name = "FemboyUI_Main",
        Parent = GuiParent,

function FemboyUI:EnableDrag(frame)
    local dragging, dragStart, startPos = false, nil, nil
    Util.Connect(frame.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = self.MainFrame.Position
            Util.Connect(input.Changed, function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    local dragInput
    Util.Connect(frame.InputChanged, function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    Util.Connect(UserInputService.InputChanged, function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            Util.Tween(self.MainFrame, {
                Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            }, 0.06, Enum.EasingStyle.Quad)
        end
    end)
end

function FemboyUI:SelectTab(tabData)
    if self.ActiveTab then
        local old = self.ActiveTab
        old.Page.Visible = false
        Util.Tween(old.Button, {BackgroundTransparency = 0.75}, 0.3)
        Util.Tween(old.Indicator, {BackgroundTransparency = 1}, 0.3)
        Util.Tween(old.Label, {TextColor3 = CurrentTheme.TextDim}, 0.3)
    end
    self.ActiveTab = tabData
    tabData.Page.Visible = true
    Util.Spring(tabData.Button, {BackgroundTransparency = 0.15}, 0.4)
    Util.Spring(tabData.Indicator, {BackgroundTransparency = 0}, 0.4)
    Util.Tween(tabData.Label, {TextColor3 = CurrentTheme.Text}, 0.3)
end

function FemboyUI:MakeToggle(tabData, config)
    config = config or {}
    local name = config.Name or "Toggle"
    local default = config.Default or false
    local callback = config.Callback or function() end
    local flag = config.Flag
    local desc = config.Description
    if flag and Config.Get(flag) ~= nil then default = Config.Get(flag) end
    local height = desc and 56 or 42
    local frame = self:MakeElementFrame(tabData, height, "Toggle_" .. name)
    local state = default
    Util.Create("TextLabel", {
        Parent = frame,
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 12, 0, desc and 5 or 0),
        Size = UDim2.new(0.65, 0, 0, desc and 22 or 42),
        Text = "🔘 " .. name,
        TextColor3 = CurrentTheme.Text,
        TextSize = 13,
        Font = Enum.Font.GothamSemibold,
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex = 63,
    })
    if desc then
        Util.Create("TextLabel", {
            Parent = frame,
            BackgroundTransparency = 1,
            Position = UDim2.new(0, 28, 0, 27),
            Size = UDim2.new(0.65, 0, 0, 16),
            Text = desc,
            TextColor3 = CurrentTheme.TextMuted,
            TextSize = 10,
            Font = Enum.Font.Gotham,
            TextXAlignment = Enum.TextXAlignment.Left,
            ZIndex = 63,
        })
    end
    local switchBg = Util.Create("Frame", {
        Parent = frame,
        BackgroundColor3 = state and CurrentTheme.Primary or CurrentTheme.SurfaceAlt,
        Size = UDim2.new(0, 46, 0, 24),
        Position = UDim2.new(1, -58, 0.5, 0),
        AnchorPoint = Vector2.new(0, 0.5),
        ZIndex = 64,
    }, {
        Util.Create("UICorner", {CornerRadius = UDim.new(1, 0)})
    })
    local knob = Util.Create("Frame", {
        Parent = switchBg,
        BackgroundColor3 = Color3.new(1, 1, 1),
        Size = UDim2.new(0, 20, 0, 20),
        Position = state and UDim2.new(1, -22, 0.5, 0) or UDim2.new(0, 2, 0.5, 0),
        AnchorPoint = Vector2.new(0, 0.5),
        ZIndex = 65,
    }, {
        Util.Create("UICorner", {CornerRadius = UDim.new(1, 0)})
    })
    local knobGlow = Util.Create("UIStroke", {
        Parent = knob,
        Color = CurrentTheme.Primary,
        Thickness = state and 2 or 0,
        Transparency = state and 0.3 or 1,
    })
    local function toggle()
        state = not state
        if state then
            Util.Spring(knob, {Position = UDim2.new(1, -22, 0.5, 0)}, 0.4)
            Util.Tween(switchBg, {BackgroundColor3 = CurrentTheme.Primary}, 0.3)
            Util.Tween(knobGlow, {Thickness = 2, Transparency = 0.3}, 0.3)
        else
            Util.Spring(knob, {Position = UDim2.new(0, 2, 0.5, 0)}, 0.4)
            Util.Tween(switchBg, {BackgroundColor3 = CurrentTheme.SurfaceAlt}, 0.3)
            Util.Tween(knobGlow, {Thickness = 0, Transparency = 1}, 0.3)
        end
        if flag then Config.Set(flag, state) end
        pcall(callback, state)
    end
    local clickArea = Util.Create("TextButton", {

function FemboyUI:MakeSlider(tabData, config)
    config = config or {}
    local name = config.Name or "Slider"
    local min = config.Min or 0
    local max = config.Max or 100
    local default = config.Default or min
    local increment = config.Increment or 1
    local callback = config.Callback or function() end
    local flag = config.Flag
    local suffix = config.Suffix or ""
    if flag and Config.Get(flag) ~= nil then default = Config.Get(flag) end
    local frame = self:MakeElementFrame(tabData, 58, "Slider_" .. name)
    local value = default
    Util.Create("TextLabel", {
        Parent = frame,
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 12, 0, 4),
        Size = UDim2.new(0.6, 0, 0, 20),
        Text = "📊 " .. name,
        TextColor3 = CurrentTheme.Text,
        TextSize = 13,
        Font = Enum.Font.GothamSemibold,
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex = 63,
    })
    local valLabel = Util.Create("TextLabel", {
        Parent = frame,
        BackgroundTransparency = 1,
        Position = UDim2.new(0.65, 0, 0, 4),
        Size = UDim2.new(0.33, 0, 0, 20),
        Text = tostring(value) .. suffix,
        TextColor3 = CurrentTheme.Primary,
        TextSize = 13,
        Font = Enum.Font.GothamBold,
        TextXAlignment = Enum.TextXAlignment.Right,
        ZIndex = 63,
    })
    local track = Util.Create("Frame", {
        Parent = frame,
        BackgroundColor3 = CurrentTheme.SurfaceAlt,
        Size = UDim2.new(1, -24, 0, 8),
        Position = UDim2.new(0, 12, 0, 36),
        ZIndex = 63,
    }, {
        Util.Create("UICorner", {CornerRadius = UDim.new(1, 0)})
    })
    local pct = (default - min) / (max - min)
    local fill = Util.Create("Frame", {
        Parent = track,
        BackgroundColor3 = CurrentTheme.Primary,
        Size = UDim2.new(pct, 0, 1, 0),
        ZIndex = 64,
    }, {
        Util.Create("UICorner", {CornerRadius = UDim.new(1, 0)})
    })
    Rainbow.AnimatedGradient(fill)
    local knobFrame = Util.Create("Frame", {
        Parent = track,
        BackgroundColor3 = Color3.new(1, 1, 1),
        Size = UDim2.new(0, 16, 0, 16),
        Position = UDim2.new(pct, 0, 0.5, 0),
        AnchorPoint = Vector2.new(0.5, 0.5),
        ZIndex = 65,
    }, {
        Util.Create("UICorner", {CornerRadius = UDim.new(1, 0)}),
        Util.Create("UIStroke", {Color = CurrentTheme.Primary, Thickness = 2, Transparency = 0.3})
    })
    local sliding = false
    local function updateSlider(inputObj)
        local trackX = track.AbsolutePosition.X
        local trackW = track.AbsoluteSize.X
        local mouseX = inputObj.Position.X
        local p = math.clamp((mouseX - trackX) / trackW, 0, 1)
        local raw = min + (max - min) * p
        local snapped = math.floor(raw / increment + 0.5) * increment
        snapped = math.clamp(snapped, min, max)
        if increment >= 1 then snapped = math.floor(snapped) else snapped = tonumber(string.format("%.2f", snapped)) end
        value = snapped
        local newP = (snapped - min) / (max - min)
        Util.Tween(fill, {Size = UDim2.new(newP, 0, 1, 0)}, 0.06, Enum.EasingStyle.Quad)
        Util.Tween(knobFrame, {Position = UDim2.new(newP, 0, 0.5, 0)}, 0.06, Enum.EasingStyle.Quad)
        valLabel.Text = tostring(snapped) .. suffix
        if flag then Config.Set(flag, snapped) end
        pcall(callback, snapped)
    end
    local inputArea = Util.Create("TextButton", {
        Parent = track,
        BackgroundTransparency = 1,
        Size = UDim2.new(1, 12, 1, 22),
        Position = UDim2.new(0, -6, 0, -11),
        Text = "",
        ZIndex = 66,
    })
    Util.Connect(inputArea.InputBegan, function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then return end
        sliding = true
        updateSlider(input.UserInputType == Enum.UserInputType.Touch and input or {Position = Vector3.new(Mouse.X, Mouse.Y, 0)})
    end)
    Util.Connect(UserInputService.InputEnded, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then sliding = false end
    end)
    Util.Connect(UserInputService.InputChanged, function(input)
        if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateSlider(input)
        end
    end)
    pcall(callback, default)
    local obj = {}
    function obj:Set(v)
        value = math.clamp(v, min, max)
        local np = (value - min) / (max - min)
        Util.Tween(fill, {Size = UDim2.new(np, 0, 1, 0)}, 0.3)
        Util.Tween(knobFrame, {Position = UDim2.new(np, 0, 0.5, 0)}, 0.3)
        valLabel.Text = tostring(value) .. suffix
        if flag then Config.Set(flag, value) end
        pcall(callback, value)
    end
    function obj:Get() return value end
    return obj
end

            Parent = optContainer,
            BackgroundColor3 = CurrentTheme.SurfaceAlt,
            BackgroundTransparency = 0.5,
            Size = UDim2.new(1, 0, 0, 28),
            Text = "  " .. tostring(optText),
            TextColor3 = CurrentTheme.Text,

            Font = Enum.Font.Gotham,

            Util.Tween(optBtn, {BackgroundTransparency = 0.5, BackgroundColor3 = CurrentTheme.SurfaceAlt}, 0.2)

    function obj:Refresh(newOpts)
        for _, c in ipairs(optContainer:GetChildren()) do if c:IsA("TextButton") then c:Destroy() end end
        options = newOpts
        for _, o in ipairs(options) do makeOption(o) end
    end
    return obj
end

function FemboyUI:MakeParagraph(tabData, config)
    config = config or {}
    local title = config.Title or "Info"
    local content = config.Content or ""
    local frame = self:MakeElementFrame(tabData, 72, "Para_" .. title)
    Util.Create("TextLabel", {
        Parent = frame,
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 12, 0, 5),
        Size = UDim2.new(1, -24, 0, 20),
        Text = "📄 " .. title,
        TextColor3 = CurrentTheme.Text,
        TextSize = 14,
        Font = Enum.Font.GothamBold,
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex = 63,
    })
    local contentLabel = Util.Create("TextLabel", {
        Parent = frame,
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 12, 0, 27),
        Size = UDim2.new(1, -24, 0, 38),
        Text = content,
        TextColor3 = CurrentTheme.TextDim,
        TextSize = 11,
        Font = Enum.Font.Gotham,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Top,
        TextWrapped = true,
        ZIndex = 63,
    })
    local obj = {}
    function obj:Set(t, c)
        if t then frame:FindFirstChild("TextLabel").Text = "📄 " .. t end
        if c then contentLabel.Text = c end
    end
    return obj
end
function FemboyUI:ToggleVisible()
    self.Visible = not self.Visible
    if self.Visible then
        self.ScreenGui.Enabled = true
        if self.MobileToggle then self.MobileToggle.Visible = self.IsMobile end
        Sound.Open()
        Util.Spring(self.MainFrame, {Size = self.Minimized and UDim2.new(0, self:GetResponsiveSize().X.Offset, 0, 48) or self:GetResponsiveSize()}, 0.55)
    else
        Sound.Close()
        Util.Tween(self.MainFrame, {Size = UDim2.new(0, 0, 0, 0)}, 0.4, Enum.EasingStyle.Back, Enum.EasingDirection.In)
        if self.MobileToggle then self.MobileToggle.Visible = self.IsMobile end
    end
end
function FemboyUI:ToggleMinimize()
    self.Minimized = not self.Minimized
    if self.Minimized then
        Util.Tween(self.Content, {Size = UDim2.new(1, 0, 0, 0)}, 0.3)
        Util.Spring(self.MainFrame, {Size = UDim2.new(0, self:GetResponsiveSize().X.Offset, 0, 48)}, 0.4)
    else
        Util.Spring(self.MainFrame, {Size = self:GetResponsiveSize()}, 0.5)
        task.delay(0.2, function()
            Util.Tween(self.Content, {Size = UDim2.new(1, 0, 1, -48)}, 0.3)
        end)
    end
end
function FemboyUI:SetTheme(themeName)
    if Themes[themeName] then
        CurrentTheme = Themes[themeName]
        Notify.Send({
            Title = "🎨 Theme Changed",
            Message = "Switched to " .. themeName .. " theme! Restart UI for full effect.",
            Type = "Info",
            Duration = 3,
        })
    end
end
function FemboyUI:Notify(config)
    Notify.Send(config)
end
function FemboyUI:Destroy()
    Util.Tween(self.MainFrame, {Size = UDim2.new(0, 0, 0, 0), BackgroundTransparency = 1}, 0.5, Enum.EasingStyle.Back, Enum.EasingDirection.In)
    Config.Save()
    task.delay(0.55, function()
        for _, conn in ipairs(AllConnections) do pcall(function() conn:Disconnect() end) end
        if self.ScreenGui then self.ScreenGui:Destroy() end
    end)
end
